cmake_minimum_required(VERSION 3.20)

project(robot_controller
    VERSION 0.1
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(robot_controller
    src/main.cpp
    src/core/Joint.cpp
    src/core/Robot.cpp
    src/control/PDController.cpp
    src/safety/SafetyLayer.cpp
    src/motor/SimulatedMotor.cpp
    src/simulation/RobotSimulator.cpp
    src/control/ControlLoop.cpp
    src/synchronization/SnapshotBuffer.cpp
    src/scheduler/PortableControlScheduler.cpp
    src/scheduler/WindowsControlScheduler.cpp
    src/scheduler/ControlSchedulerFactory.cpp
    src/scheduler/LinuxControlScheduler.cpp
    src/communication/CanFrame.cpp
    src/communication/SimulatedCan.cpp
    src/communication/JointCommandCodec.cpp
    src/communication/CanCommandBuffer.cpp
    src/communication/CanStateBuffer.cpp
    src/communication/CanCommunication.cpp
    src/communication/JointStateCodec.cpp
)

target_include_directories(robot_controller
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)
